public class ResponseVO {
    private  Integer code;
    private String status;
    private String message;
    private Object data;

    public ResponseVO() {
    }

    public ResponseVO(Integer code, String status, String message, Object data) {
        this.code = code;
        this.status = status;
        this.message = message;
        this.data = data;
    }

    public static ResponseVO success(Object data) {
        return new ResponseVO(
                ResponseCodeEnum.SUCCESS.getCode(),
                "success",
                ResponseCodeEnum.SUCCESS.getMessage(),
                data
        );
    }

    public static ResponseVO fail(ResponseCodeEnum codeEnum) {
        return new ResponseVO(
                codeEnum.getCode(),
                "fail",
                codeEnum.getMessage(),
                null
        );
    }

    public static ResponseVO fail(Integer code, String message) {
        return new ResponseVO(
                code,
                "fail",
                message,
                null
        );
    }

    // ---------------- getter / setter ----------------

    public Integer getCode() {
        return code;
    }

    public void setCode(Integer code) {
        this.code = code;
    }

    public Object getData() {
        return data;
    }

    public void setData(Object data) {
        this.data = data;
    }

    public String getMessage() {
        return message;
    }

    public void setMessage(String message) {
        this.message = message;
    }

    public String getStatus() {
        return status;
    }

    public void setStatus(String status) {
        this.status = status;
    }
}
